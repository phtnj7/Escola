Estruturas de Dados — Exemplos

1. Array

const frutas = ["maçã", "banana", "laranja"];
frutas.push("uva");
console.log(frutas);
console.log("Primeira:", frutas[0]);

2. Lista Ligada Simples

const a = new No(1);
const b = new No(2);
const c = new No(3);
a.proximo = b;
b.proximo = c;

let atual = a;
while (atual != null) {
  console.log(atual.valor);
  atual = atual.proximo;
}

3. Lista Duplamente Ligada

const n1 = new NoDuplo(10);
const n2 = new NoDuplo(20);
const n3 = new NoDuplo(30);
n1.proximo = n2; n2.anterior = n1;
n2.proximo = n3; n3.anterior = n2;

let cur = n1;
while (cur) { console.log("→", cur.valor); cur = cur.proximo; }
cur = n3;
while (cur) { console.log("←", cur.valor); cur = cur.anterior; }

4. Lista Circular

const x = new No(1);
const y = new No(2);
const z = new No(3);
x.proximo = y; y.proximo = z;
z.proximo = x; // volta pro início

let n = x;
for (let i = 0; i < 6; i++) {
  console.log(n.valor);
  n = n.proximo;
}

5. Pilha (Stack)

const pilha = [];
pilha.push(10); pilha.push(20); pilha.push(30);
console.log("Pilha:", pilha);
console.log("Removido:", pilha.pop());
console.log("Pilha agora:", pilha);

6. Fila (Queue)

const fila = [];
fila.push("A"); fila.push("B"); fila.push("C");
console.log("Fila:", fila);
console.log("Removido:", fila.shift());
console.log("Fila agora:", fila);

7. Alocação Dinâmica

let pessoa = { nome: "João", idade: 20 };
console.log(pessoa);
pessoa = null;
console.log("Após liberar:", pessoa);

8. Recursividade

function fatorial(n) {
  if (n == 0) return 1;
  return n * fatorial(n - 1);
}
console.log("4! =", fatorial(4));
console.log("5! =", fatorial(5));

9. Lista Encadeada (com classe)

const lista = new Lista();
lista.adicionar(1);
lista.adicionar(2);
lista.adicionar(3);
lista.mostrar();

10. Tabela Hash

const tabela = {};
tabela["nome"] = "Maria";
tabela["idade"] = 22;
tabela["curso"] = "TI";
console.log(tabela["nome"]);
console.log(tabela["idade"]);
console.log(tabela["curso"]);

12. Ordenação (Bubble Sort)

console.log("Antes:", [5, 3, 8, 1, 4]);
console.log("Depois:", bubbleSort([5, 3, 8, 1, 4]));

13. Busca Linear e Binária

const numeros = [10, 20, 30, 40, 50];
console.log("Linear — buscar 30:", buscaLinear(numeros, 30));
